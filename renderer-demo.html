🎞️ Canvas Renderer — Demo

Chọn khung + tải ảnh → xuất PNG chất lượng cao

// Cách dùng trong code
import { PhotoBoothRenderer, FRAME_CONFIGS } from './photobooth-renderer.js';

const renderer = new PhotoBoothRenderer();

// Render:
const dataURL = await renderer.render(
  photosArray,          // string[] — data URLs or src URLs
  FRAME_CONFIGS.pink_ribbon,
  { dateText: '2026.09.28' }
);

// Hiển thị preview:
previewImg.src = dataURL;

// Tải về:
await renderer.download(
  photosArray,
  FRAME_CONFIGS.monchhichi
);
